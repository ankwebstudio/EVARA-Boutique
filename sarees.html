document.addEventListener("DOMContentLoaded", () => { let cart = JSON.parse(localStorage.getItem("evaraCart")) || []; let wishlist = JSON.parse(localStorage.getItem("evaraWishlist")) || []; let activePanel = "cart"; // SAVE DATA function saveData() { localStorage.setItem("evaraCart", JSON.stringify(cart)); localStorage.setItem("evaraWishlist", JSON.stringify(wishlist)); updateCounts(); } // GET PRODUCT DETAILS function getProduct(card) { return { id: card.querySelector("h3").textContent.trim(), name: card.querySelector("h3").textContent.trim(), price: Number( card.querySelector(".saree-bottom strong") .textContent.replace(/[^\d]/g, "") ), image: card.querySelector("img").getAttribute("src") }; } // ADD TO CART function addToCart(product) { let existing = cart.find(item => item.id === product.id); if (existing) { existing.qty++; } else { cart.push({...product, qty: 1}); } saveData(); openSidebar("cart"); } // ADD / REMOVE WISHLIST function toggleWishlist(product) { let existing = wishlist.find(item => item.id === product.id); if (existing) { wishlist = wishlist.filter(item => item.id !== product.id); } else { wishlist.push(product); } saveData(); updateWishlistIcons(); openSidebar("wishlist"); } // UPDATE HEADER COUNTS function updateCounts() { const cartCount = cart.reduce( (total, item) => total + item.qty, 0 ); document.getElementById("cartCount").textContent = cartCount; document.getElementById("wishlistCount").textContent = wishlist.length; } // OPEN SIDEBAR function openSidebar(type) { activePanel = type; document.getElementById("shopSidePanel") .classList.add("is-open"); document.getElementById("shopPanelOverlay") .classList.add("is-visible"); document.body.classList.add("shop-panel-open"); renderSidebar(); } // CLOSE SIDEBAR function closeSidebar() { document.getElementById("shopSidePanel") .classList.remove("is-open"); document.getElementById("shopPanelOverlay") .classList.remove("is-visible"); document.body.classList.remove("shop-panel-open"); } // RENDER CART / WISHLIST function renderSidebar() { const container = document.getElementById("shopPanelItems"); const isCart = activePanel === "cart"; const products = isCart ? cart : wishlist; document.getElementById("shopPanelTitle").textContent = isCart ? "Shopping Bag" : "Your Wishlist"; document.getElementById("shopPanelFooter").style.display = isCart ? "block" : "none"; document.querySelectorAll(".shop-tab").forEach(tab => { tab.classList.toggle( "active", tab.dataset.panel === activePanel ); }); if (products.length === 0) { container.innerHTML = `

${isCart ? "Your bag is empty" : "Your wishlist is empty"}

Discover something beautiful from EVARA.

`; } else { container.innerHTML = products.map(item => `

${item.name}

₹${item.price.toLocaleString("en-IN")}

${isCart ? `
${item.qty}
` : ` `}
`).join(""); } updateSubtotal(); } // UPDATE SUBTOTAL function updateSubtotal() { let total = cart.reduce( (sum, item) => sum + item.price * item.qty, 0 ); document.getElementById("cartSubtotal").textContent = "₹" + total.toLocaleString("en-IN"); document.getElementById("cartTabCount").textContent = cart.reduce((sum, item) => sum + item.qty, 0); document.getElementById("wishlistTabCount").textContent = wishlist.length; document.getElementById("shopCheckout").disabled = cart.length === 0; } // UPDATE HEART ICONS function updateWishlistIcons() { document.querySelectorAll(".saree-product").forEach(card => { const product = getProduct(card); const button = card.querySelector(".saree-wishlist"); const active = wishlist.some( item => item.id === product.id ); button.innerHTML = active ? '' : ''; button.classList.toggle("is-active", active); }); } // PRODUCT BUTTON EVENTS document.querySelectorAll(".saree-product").forEach(card => { const product = getProduct(card); card.querySelector(".saree-wishlist") .addEventListener("click", () => { toggleWishlist(getProduct(card)); }); card.querySelector(".saree-add-cart") .addEventListener("click", () => { addToCart(getProduct(card)); }); }); // SIDEBAR ITEM ACTIONS document.getElementById("shopPanelItems") .addEventListener("click", event => { const button = event.target.closest("[data-action]"); if (!button) return; const id = button.dataset.id; const action = button.dataset.action; if (action === "increase") { const item = cart.find(p => p.id === id); if (item) item.qty++; } if (action === "decrease") { const item = cart.find(p => p.id === id); if (item) { item.qty--; if (item.qty <= 0) { cart = cart.filter(p => p.id !== id); } } } if (action === "remove") { if (activePanel === "cart") { cart = cart.filter(p => p.id !== id); } else { wishlist = wishlist.filter(p => p.id !== id); } } if (action === "move") { const item = wishlist.find(p => p.id === id); if (item) { const existing = cart.find(p => p.id === id); if (existing) { existing.qty++; } else { cart.push({...item, qty: 1}); } wishlist = wishlist.filter(p => p.id !== id); } } saveData(); renderSidebar(); updateWishlistIcons(); }); // HEADER ICON EVENTS document.querySelectorAll(".js-open-panel").forEach(button => { button.addEventListener("click", () => { openSidebar(button.dataset.panel); }); }); // SIDEBAR TABS document.querySelectorAll(".shop-tab[data-panel]") .forEach(button => { button.addEventListener("click", () => { openSidebar(button.dataset.panel); }); }); // CLOSE BUTTON document.getElementById("shopPanelClose") .addEventListener("click", closeSidebar); document.getElementById("shopPanelOverlay") .addEventListener("click", closeSidebar); document.addEventListener("keydown", event => { if (event.key === "Escape") closeSidebar(); }); // INITIALIZE updateCounts(); updateWishlistIcons(); });
